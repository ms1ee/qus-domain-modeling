I made some interpretation choices where the PRD and Interface leave details open. The code has not been run.

- **Custom list separator:** custom list lines accept `,`, `;`, or the full-width `，` and `；` as separators.
- **Default prizes:** `load_default_prizes` adds the preset prizes and returns a list of errors. The drawers for the three presets are Chen Qixian, Jin Duo and Ma Jiaqi.
- **Fallback rule:** `last_two_draw_results` holds booleans recording whether each of the last two Chen Qixian / Jin Duo rounds included a Cheng Ru group member. When the drawer is Ma Jiaqi and the list holds two `False` values, the pool is limited to Cheng Ru group members, ignoring avoidance groups but still excluding earlier winners.
- **`_get_eligible_pool`:** it excludes previous winners and avoidance groups. For drawer "Li Bin" it applies only the winner exclusion. The ordering rule (avoidance-group members eligible but drawn after others) is left to the draw loop, which is why `draw_for_prize` appends them after the main pool.
- **Pause/continue/terminate:** these are read from a console command between each winner draw.
- **Winner tracking:** `LotteryManager` keeps its own winner set, and `record_winner` is called when someone wins, so winner exclusion works even without a `ResultManager`.
- **Prize ordering for the summary:** the summary lists prizes in the current configured order (treated as high to low), with winners in winning-time order within each prize.

from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Product Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Cheng Ru group"),
    ("Zhou Ba", "Cheng Ru group"),
    ("Wu Jiu", "Marketing Department"),
    ("Zheng Shi", "Marketing Department"),
    ("Chen Qixian", "Technology Department"),
    ("Jin Duo", "Product Department"),
    ("Ma Jiaqi", "Marketing Department"),
    ("Li Bin", "Operations Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Error reading file: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = re.split(r"[,;，；]", line)
            parts = [p.strip() for p in parts]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = OrderedDict()
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self._prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3, "Chen Qixian"),
            ("Second Prize - Shopping Cards", 2, "Jin Duo"),
            ("First Prize - 1 Day Annual Leave", 1, "Ma Jiaqi"),
        ]
        for n, q, d in defaults:
            try:
                self.add_prize(n, q, None, d)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or int(quantity) <= 0:
            raise ValueError("Invalid prize")
        p = Prize(name, int(quantity), list(avoid_groups or []), associated_drawer)
        self._prizes.append(p)
        return p

    def get_prizes(self):
        return list(self._prizes)

    def reorder_prizes(self, order):
        n = len(self._prizes)
        if len(order) != n or sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize, time=None):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        self.winners.append(Winner(user, prize))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = sorted(
                [w for w in self.winners if w.prize.name == p.name],
                key=lambda w: w.time,
            )
        return result

    def export_to_csv(self, prizes):
        try:
            path = os.path.join(
                os.getcwd(), f"awards_result_{datetime.now():%Y%m%d}.txt")
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for p in prizes:
                    for w in grouped.get(p.name, []):
                        f.write(f"{w.user.name},{w.user.department},{p.name},"
                                f"{w.time:%Y-%m-%d %H:%M:%S}\n")
            return True, f"Results successfully exported to {path}"
        except OSError as e:
            return False, f"Export failed: {e}"

import random

CHENG_RU = "Cheng Ru group"
FALLBACK_SOURCES = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET = "Ma Jiaqi"
SPECIAL = "Li Bin"


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def record_winner(self, user):
        if user not in self.winners:
            self.winners.append(user)

    def _available(self):
        names = {id(w) for w in self.winners}
        return [u for u in self.users if id(u) not in names
                and u not in self.winners]

    def _get_eligible_pool(self, prize):
        pool = self._available()
        drawer = prize.associated_drawer
        if drawer == FALLBACK_TARGET and len(self.last_two_draw_results) >= 2 \
                and not any(self.last_two_draw_results[-2:]):
            return [u for u in pool if u.department == CHENG_RU]
        if drawer == SPECIAL:
            return pool
        return [u for u in pool if u.department not in prize.avoid_groups]

    def _avoided_pool(self, prize):
        if prize.associated_drawer == SPECIAL:
            return []
        return [u for u in self._available()
                if u.department in prize.avoid_groups]

    def draw_for_prize(self, prize, on_draw=None, control=None):
        """Draw winners; yields list of users. control() may return
        'pause'/'continue'/'terminate'."""
        drawn = []
        main = self._get_eligible_pool(prize)
        fallback_active = (prize.associated_drawer == FALLBACK_TARGET
                           and len(self.last_two_draw_results) >= 2
                           and not any(self.last_two_draw_results[-2:]))
        random.shuffle(main)
        extra = [] if fallback_active else self._avoided_pool(prize)
        random.shuffle(extra)
        queue = main + extra
        for i in range(prize.quantity):
            if control:
                if control() == "terminate":
                    break
            if not queue:
                break
            u = queue.pop(0)
            self.record_winner(u)
            drawn.append(u)
            if on_draw:
                on_draw(i + 1, u)
        if prize.associated_drawer in FALLBACK_SOURCES:
            self.last_two_draw_results.append(
                any(u.department == CHENG_RU for u in drawn))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        return drawn

    for i, p in enumerate(pm.get_prizes(), 1):
        print(i, p.name, p.quantity)
    o = input("New order (space separated, blank to keep): ").split()
    if o:
        try:
            if not pm.reorder_prizes([int(x) for x in o]):
                print("Invalid order, keeping default")
        except ValueError:
            print("Invalid order")
    lot = LotteryManager(users)
    rm = ResultManager()
    state = {"paused": False}

    def control():
        return None

    for p in pm.get_prizes():
        print(f"\n== {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")

        if cmd == "terminate":

                break
